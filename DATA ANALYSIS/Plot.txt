With datasets 
import seaborn as sns
import matplotlib.pyplot as plt

flights = sns.load_dataset("flights")

flights = flights.pivot(
    index="month",
    columns="year",
    values="passengers"
)

sns.heatmap(flights)
plt.show()

To get value in integer - heat_map_5 = sns.heatmap(flights, annot = True, fmt = "d")
To get value in boolean - heat_map_5 = sns.heatmap(flights, annot = True, fmt = ".1f")
To change the color scheme - heat_map_6 = sns.heatmap(flights, cmap = 'YlGnBu')
To remove the side bar - heat_map_8 = sns.heatmap(flights, cbar = False)
